```python
@classmethod
def validate_rate_limit(cls, v: str) -> str:
    # Parses "100/minute", "1000/hour", etc.
    if not re.match(r"^\d+/(second|minute|hour|day)$", v):
        raise ValueError("Invalid rate limit format")
    return v

@field_validator("CORS_ORIGINS", mode="before")
@classmethod
def parse_cors_origins(cls, v: Union[str, List[str]]) -> List[str]:
    if isinstance(v, str):
        return [origin.strip() for origin in v.split(",")]
    return v
```

### 5.2 Recommended: Hot Reload for Safe Settings
```python
# backend/config.py
class Settings(BaseSettings):
    # ...
    
    # These can be hot-reloaded
    HOT_RELOADABLE = {
        "LOG_LEVEL",
        "RATE_LIMIT_DEFAULT",
        "RATE_LIMIT_AUTHENTICATED",
        "RATE_LIMIT_PREMIUM",
        "RAG_TOP_K",
        "RAG_SIMILARITY_THRESHOLD",
        "WEB_SEARCH_ENABLED",
        "OLLAMA_AUTO_START",
    }
    
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self._callbacks: Dict[str, List[Callable]] = defaultdict(list)
    
    def on_change(self, key: str, callback: Callable) -> None:
        self._callbacks[key].append(callback)
    
    def __setattr__(self, name: str, value: Any) -> None:
        old = getattr(self, name, None)
        super().__setattr__(name, value)
        if name in self.HOT_RELOADABLE and old != value:
            for cb in self._callbacks[name]:
                try:
                    cb(value)
                except Exception:
                    logger.exception(f"Config callback failed for {name}")

# Usage
settings = get_settings()
settings.on_change("LOG_LEVEL", lambda v: logging.getLogger().setLevel(v))
settings.on_change("RATE_LIMIT_DEFAULT", lambda v: rate_limiter.update_default(v))
```

### 7.1 Required: Drift Detection
```python
# backend/config_drift.py
class ConfigDriftDetector:
    def __init__(self, settings: Settings):
        self.settings = settings
        self.baseline: Dict[str, Any] = {}
    
    def snapshot(self) -> Dict[str, Any]:
        """Capture current effective config."""
        return {
            key: getattr(self.settings, key)
            for key in self.settings.model_fields
            if not key.startswith("_")
        }
    
    def detect_drift(self, current: Dict[str, Any]) -> List[Drift]:
        """Compare against baseline."""
        drifts = []
        for key, expected in self.baseline.items():
            actual = current.get(key)
            if actual != expected:
                drifts.append(Drift(
                    key=key,
                    expected=expected,
                    actual=actual,
                    severity=self._severity(key),
                    detected_at=datetime.utcnow(),
                ))
        return drifts
    
    def _severity(self, key: str) -> DriftSeverity:
        critical = {"MASTER_KEY", "SECRET_KEY", "DATABASE_URL", "REDIS_URL"}
        if key in critical:
            return DriftSeverity.CRITICAL
        return DriftSeverity.WARNING
```

```python
        "RATE_LIMIT_PREMIUM": "500/minute",
    },

async def migrate_config(current_version: str, settings: Settings) -> Settings:
    for version in sorted(MIGRATIONS.keys()):
        if version > current_version:
            migrations = MIGRATIONS[version](settings)
            for key, value in migrations.items():
                setattr(settings, key, value)
    return settings
```
